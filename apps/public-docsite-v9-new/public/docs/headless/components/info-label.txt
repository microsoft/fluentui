# Info Label

## Examples

### Default

```tsx
import * as React from "react";
import { InfoLabel } from "@fluentui/react-headless-components-preview/info-label";
import { Info16Regular } from "@fluentui/react-icons";

import styles from "./styles/info-label.module.css";

export const Default = (): React.ReactNode => (
  <InfoLabel
    className={styles.infoLabel}
    info={{
      className: styles.popover,
      children: (
        <>
          This is an InfoLabel's information.
          <a href="https://storybooks.fluentui.dev/headless/">Learn more</a>
        </>
      ),
    }}
    infoButton={{
      className: styles.infoButton,
      children: <Info16Regular />,
    }}
  >
    Example label
  </InfoLabel>
);
```

### InField

```tsx
import * as React from "react";
import { Info16Regular } from "@fluentui/react-icons";
import { Field } from "@fluentui/react-headless-components-preview/field";
import { InfoLabel } from "@fluentui/react-headless-components-preview/info-label";
import { Input } from "@fluentui/react-headless-components-preview/input";
import type { LabelProps } from "@fluentui/react-headless-components-preview/label";

import fieldStyles from "./styles/field.module.css";
import infoLabelStyles from "./styles/info-label.module.css";
import inputStyles from "./styles/input.module.css";

export const InField = (): React.ReactNode => (
  <Field
    className={fieldStyles.field}
    label={{
      children: (_: unknown, props: LabelProps) => (
        <InfoLabel
          {...props}
          className={infoLabelStyles.infoLabel}
          info={{
            className: infoLabelStyles.popover,
            children: "Example info",
          }}
          infoButton={{
            className: infoLabelStyles.infoButton,
            children: <Info16Regular />,
          }}
        >
          Field with info label
        </InfoLabel>
      ),
    }}
  >
    <Input
      className={inputStyles.wrap}
      input={{ className: inputStyles.input }}
      placeholder="Default input"
    />
  </Field>
);
```
